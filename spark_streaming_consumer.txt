"""
Materia: Procesamiento de Datos (Big Data) - UNAD
Tarea 4: Análisis de Datos en Tiempo Real con Apache Kafka y Spark Streaming
Descripción: Script Consumidor que se conecta a Kafka usando Spark Structured Streaming,
lee el flujo de datos en tiempo real, aplica un esquema y calcula promedios 
por ventanas de tiempo de 1 minuto.
"""

from pyspark.sql import SparkSession
from pyspark.sql.functions import from_json, col, window
from pyspark.sql.types import StructType, StructField, IntegerType, FloatType, TimestampType

# 1. Crear y configurar la sesión de Spark
spark = SparkSession.builder \
    .appName("SensorDataAnalysisStreaming") \
    .getOrCreate()

# Reducir el nivel de logs molestos en la consola
spark.sparkContext.setLogLevel("WARN")

# 2. Definir el esquema estructural de los datos JSON entrantes
schema = StructType([
    StructField("sensor_id", IntegerType()),
    StructField("temperature", FloatType()),
    StructField("humidity", FloatType()),
    StructField("timestamp", TimestampType())
])

# 3. Configurar la lectura de streaming conectándose a Apache Kafka
df = spark \
    .readStream \
    .format("kafka") \
    .option("kafka.bootstrap.servers", "localhost:9092") \
    .option("subscribe", "sensor_data") \
    .load()

# 4. Parsear el campo 'value' binario recibido de Kafka y transformarlo usando el esquema definido
parsed_df = df.select(
    from_json(col("value").cast("string"), schema).alias("data")
).select("data.*")

# 5. Aplicar operaciones de análisis: Agrupar por ventana de tiempo (1 min) y ID del sensor, calculando promedios
windowed_stats = parsed_df \
    .groupBy(
        window(col("timestamp"), "1 minute"), 
        col("sensor_id")
    ) \
    .agg({
        "temperature": "avg", 
        "humidity": "avg"
    })

# 6. Escribir y mostrar los resultados en tiempo real por la consola
query = windowed_stats \
    .writeStream \
    .outputMode("complete") \
    .format("console") \
    .start()

# Mantener la aplicación de streaming ejecutándose en curso
query.awaitTermination()

"""
Materia: Procesamiento de Datos (Big Data) - UNAD
Tarea 4: Análisis de Datos en Tiempo Real con Apache Kafka y Spark Streaming
Descripción: Script Productor que simula lecturas de sensores (temperatura y humedad) 
y los transmite en tiempo real a un tópico de Apache Kafka.
"""

import time
import json
import random
from kafka import KafkaProducer

def generate_sensor_data():
    """
    Función que genera un diccionario con datos simulados de un sensor IoT.
    Retorna:
        dict: Contiene ID del sensor, temperatura, humedad y marca de tiempo.
    """
    return {
        "sensor_id": random.randint(1, 10),
        "temperature": round(random.uniform(20, 30), 2),
        "humidity": round(random.uniform(30, 70), 2),
        "timestamp": int(time.time())
    }

# Inicialización del Productor de Kafka conectado al servidor local
producer = KafkaProducer(
    bootstrap_servers=['localhost:9092'],
    value_serializer=lambda x: json.dumps(x).encode('utf-8') # Serializa los datos a formato JSON bytes
)

print("Iniciando envío de datos al tópico 'sensor_data'...")

# Bucle infinito para enviar datos continuamente cada 1 segundo
while True:
    sensor_data = generate_sensor_data()
    # Envía el mensaje al tópico 'sensor_data'
    producer.send('sensor_data', value=sensor_data)
    print(f"Enviado exitosamente: {sensor_data}")
    time.sleep(1)  # Pausa de 1 segundo entre cada envío
